# 실행 로그 연결을 쓸 때만 빌리기 (준비하기 후속)

> 📝 기준: API 명세 v0.4 2장 · PRD v0.9 8장(지표) · 2026-10-14
>
> **`재수집_서류보존_구현.md`를 먼저 합친 뒤에 넣는다.** 두 PR이 같은 테스트 파일 2개를 건드리고, 이 문서의 코드는 그 PR이 들어간 상태를 전제로 쓰여 있다

준비하기(#27) 검토에서 나온 것이다. `agent_run`이 **실행이 끝날 때까지 실행 로그용 연결 하나를 쥐고 있다.** 준비하기는 모델에 서류 안내를 묻는 동안에도 그 연결을 들고 있어서, 로그 풀(`max_size=5`)이 준비하기의 동시 처리 수를 정한다. 여섯 번째 학생이 누르면 연결을 기다리다 30초 뒤 500이 난다. 모델이 느릴수록 창이 넓어진다.

요건 추출·학교 공지 수집·청년정책 수집도 같은 구조다. 수집은 한 번에 하나만 돌아 지금은 티가 안 나지만, 로그를 쓰는 방식이 두 가지로 갈리지 않게 **호출부를 모두 같이 바꾼다**(10/14 선택).

**규모와 확인:**
- api 파일 16개가 바뀐다(앱 8개, 테스트 8개). 새 파일이 없다.
- **DB 변경이 없다.** 합친 뒤 db push가 필요 없다. web도 바뀌지 않는다.
- api `ruff check`·`ruff format --check` 통과, `552 passed, 74 skipped`(DB 없이) · `626 passed`(새 DB).
- 되돌림 시험: 옛 방식(실행 내내 연결 쥐기)으로 되돌리면 새 테스트 3개가 모두 잡는다(아래 "확인한 것").
- 테스트 수가 늘어난 것은 새 테스트 3개 때문이고, **기존 테스트의 기대값은 하나도 바꾸지 않았다**(호출하는 모양만 바꿨다).

## 남은 것

- **로그 풀 크기는 측정하고 정한다.** 지금 `max_size=5`를 그대로 뒀다. 연결을 오래 쥐지 않게 됐으니 당장은 넉넉하지만, W9–10 측정에서 동시 실행 수와 대기 시간을 보고 정한다.
- **연결을 못 얻을 때는 아직 기다린다.** 풀이 꽉 차면 psycopg 기본값대로 30초를 기다린다. 바로 `503 DB_UNAVAILABLE`로 떨구려면 `pool.connection(timeout=...)`을 공급자에 넣어야 하는데, 얼마가 맞는지는 측정하고 정한다.
- **수집은 여전히 한 번에 하나다.** 이 PR은 로그 연결만 푼다. 수집을 여러 개 동시에 돌리려면 데이터 쪽 연결과 외부 API 호출 수를 따로 봐야 한다.
- **실행 로그 쓰기가 실패하면 그 실행은 터진다.** 전과 같다(빌리는 자리가 늘어 터질 지점이 늘었을 뿐 동작은 같다). 로그를 못 써도 일은 계속하게 할지는 따로 정한다.

```python
# autocommit 연결을 주는 공급자. 풀이면 pool.connection, 풀이 없으면 using(conn)이다
LogConnect = Callable[[], AbstractAsyncContextManager[AsyncConnection]]

SECRET_HINTS = ("token", "secret", "password", "authorization", "api_key")
# 프로필 값은 LLM에도 로그에도 넣지 않는다(PRD 6장). 실수로 들어와도 여기서 지운다
PROFILE_KEYS = frozenset(PROFILE_FIELDS)
_KST = timezone(timedelta(hours=9))


def _hidden(key: str) -> bool:
    lowered = key.lower()
    return lowered in PROFILE_KEYS or any(hint in lowered for hint in SECRET_HINTS)


def scrub(value: Any) -> Any:
    """비밀값이나 프로필 항목 이름의 키는 값을 지운다. tool_calls에 남지 않게 하는 마지막 안전장치."""
    if isinstance(value, dict):
        return {
            key: "[REDACTED]" if _hidden(str(key)) else scrub(item) for key, item in value.items()
        }
    if isinstance(value, list):
        return [scrub(item) for item in value]
    return value


def _elapsed_ms(started: float) -> int:
    return int((time.perf_counter() - started) * 1000)


@dataclass
class ToolCall:
    id: str
    output: Any = None


def using(conn: AsyncConnection) -> LogConnect:
    """이미 연 autocommit 연결 하나를 그대로 쓰는 공급자. 풀이 없는 곳(CLI·테스트)에서 쓴다."""
    _require_autocommit(conn)

    @asynccontextmanager
    async def connect() -> AsyncIterator[AsyncConnection]:
        yield conn

    return connect


@dataclass
class AgentRun:
    connect: LogConnect
    id: str
    input_tokens: int = 0
    output_tokens: int = 0
    cost_usd: Decimal = Decimal(0)
    _seq: int = 0

    @asynccontextmanager
    async def tool(self, name: str, input: dict[str, Any] | None = None) -> AsyncIterator[ToolCall]:
        self._seq += 1
        started = time.perf_counter()
        async with self.connect() as conn:
            cursor = await conn.execute(
                "insert into tool_calls (run_id, seq, tool_name, input, status)"
                " values (%s, %s, %s, %s, 'running') returning id",
                (self.id, self._seq, name, Jsonb(scrub(input or {}))),
            )
            call = ToolCall(id=str((await cursor.fetchone())["id"]))
        status = "failed"
        try:
            yield call
            status = "succeeded"
        finally:
            output = Jsonb(scrub(call.output)) if call.output is not None else None
            async with self.connect() as conn:
                await conn.execute(
                    "update tool_calls set output = %s, status = %s, latency_ms = %s where id = %s",
                    (output, status, _elapsed_ms(started), call.id),
                )

    async def record_llm(
        self,
        *,
        provider: str,
        model: str,
        input_tokens: int,
        output_tokens: int,
        latency_ms: int,
        cost_usd: Decimal | float | None = None,
        tool_call: ToolCall | None = None,
        priced_on: date | None = None,
    ) -> None:
        """LLM 호출 1건. cost_usd를 빼면 priced_on(기본: 오늘 KST)의 단가로 계산한다."""
        if cost_usd is None:
            on = priced_on or datetime.now(_KST).date()
            cost = priced_cost(provider, model, input_tokens, output_tokens, on)
        else:
            cost = Decimal(str(cost_usd))
        async with self.connect() as conn:
            await conn.execute(
                "insert into llm_calls (run_id, tool_call_id, provider, model, input_tokens,"
                " output_tokens, cost_usd, latency_ms) values (%s, %s, %s, %s, %s, %s, %s, %s)",
                (
                    self.id,
                    tool_call.id if tool_call else None,
                    provider,
                    model,
                    input_tokens,
                    output_tokens,
                    cost,
                    latency_ms,
                ),
            )
        self.input_tokens += input_tokens
        self.output_tokens += output_tokens
        self.cost_usd += cost


async def start_run(
    connect: LogConnect,
    *,
    trigger: str,
    user_id: str | None = None,
    scenario_code: str | None = None,
) -> str:
    """실행 행만 만든다(status running). 이어서 agent_run(run_id=...)으로 기록하거나 abandon_run으로 닫는다."""
    async with connect() as conn:
        _require_autocommit(conn)
        cursor = await conn.execute(
            "insert into agent_runs (user_id, trigger_type, scenario_code) values (%s, %s, %s)"
            " returning id",
            (user_id, trigger, scenario_code),
        )
        return str((await cursor.fetchone())["id"])


async def abandon_run(connect: LogConnect, run_id: str, error: ApiError) -> None:
    """시작하지 않을 실행을 실패로 닫는다(같은 공고를 그새 다른 요청이 준비하기 시작했을 때 등)."""
    async with connect() as conn:
        await conn.execute(
            "update agent_runs set status = 'failed', finished_at = now(), latency_ms = 0,"
            " error_message = %s where id = %s and status = 'running'",
            (error_text(error), run_id),
        )


@asynccontextmanager
async def agent_run(
    connect: LogConnect,
    *,
    trigger: str,
    user_id: str | None = None,
    scenario_code: str | None = None,
    run_id: str | None = None,
) -> AsyncIterator[AgentRun]:
    """실행 하나를 기록한다. connect는 autocommit 연결 공급자다(db.get_log_connect, 또는 using(conn)).

    한 줄 쓸 때마다 빌렸다 돌려주므로, 모델을 기다리는 동안 로그 연결을 쥐지 않는다.
    autocommit이어야 실패한 실행도 남는다. run_id를 주면 start_run으로 만든 행을 이어서 쓴다
    (latency_ms는 여기부터 잰다).
    """
    started = time.perf_counter()
    if run_id is None:
        run_id = await start_run(
            connect, trigger=trigger, user_id=user_id, scenario_code=scenario_code
        )
    else:
        async with connect() as conn:
            _require_autocommit(conn)
    run = AgentRun(connect=connect, id=run_id)
    try:
        yield run
    except Exception as exc:
        await _finish(run, started, "failed", error_text(exc))
        raise
    await _finish(run, started, "succeeded", None)


def error_text(exc: BaseException) -> str:
    """agent_runs.error_message. ApiError는 "코드: 문구", 그 밖은 내부 기록(repr)이다."""
    if isinstance(exc, ApiError):
        return f"{exc.code}: {exc.message}"[:500]
    return repr(exc)[:500]


def _require_autocommit(conn: AsyncConnection) -> None:
    if not conn.autocommit:
        raise RuntimeError("agent_run에는 autocommit 연결 공급자(get_log_connect)를 넘겨야 해요.")


async def _finish(run: AgentRun, started: float, status: str, error: str | None) -> None:
    async with run.connect() as conn:
        await conn.execute(
            "update agent_runs set status = %s, finished_at = now(), latency_ms = %s,"
            " input_tokens = %s, output_tokens = %s, cost_usd = %s, error_message = %s"
            " where id = %s",
            (
                status,
                _elapsed_ms(started),
                run.input_tokens,
                run.output_tokens,
                run.cost_usd,
                error,
                run.id,
            ),
        )
```

```python
def check(paths: Sequence[Path]) -> int:
    """폴더마다 읽은 값과 첨부 짝을 찍고, 끝에 첨부 형식을 센다. 고칠 것이 있으면 1을 돌려준다."""
    folders = _folders(paths)
    if folders is None:
        return 1
    problems = _duplicates(folders)
    formats: Counter[str] = Counter()
    plan: Counter[str] = Counter()
    for no, folder in enumerate(folders, 1):
        print(f"[{no}/{len(folders)}] {folder.name}")
        saved = _load(folder, problems)
        if isinstance(saved, str):
            print(f"    {_FIX}: {saved}")
            plan[_FIX] += 1
            continue
        page = saved.page
        print(f"    {page.title}")
        print(
            f"    글 {page.entry_id} · {page.posted_on or '등록일 모름'}"
            f" · {page.category or '분류 없음'}({page.opportunity_category})"
            f" · 캠퍼스 {page.campus or '없음'} → {_SCOPE[page.scope]}"
        )
        for line, label in _attachment_lines(saved):
            print(f"    {line}")
            formats[label] += 1
        for note in _notes(saved):
            print(f"    {note}")
        if page.scope == "erica" and saved.missing:
            print(
                f"    {_FIX}: 가져오지 못한 파일을 고친다(없이 가져오려면 import --allow-missing)"
            )
            plan[_FIX] += 1
        else:
            plan[_SCOPE[page.scope]] += 1
    print()
    print("공지 " + " · ".join(f"{name} {count}" for name, count in plan.items()))
    if formats:
        print(f"첨부 형식(본문 이미지 포함 {sum(formats.values())}개)")
        for label, count in formats.most_common():
            print(f"    {label}: {count}")
    return 1 if plan[_FIX] else 0


def _attachment_lines(saved: SavedNotice) -> list[tuple[str, str]]:
    """(첨부 한 줄, 형식 집계 이름). 한 줄은 모델이 첫 메시지에서 보는 첨부 목록과 같다."""
    library = Library(saved.notice_input(), Limits())
    listing = dict(zip(library.items, library.listing(), strict=True))
    lines = []
    for file in saved.files:
        prefix = "본문 이미지 " if file.body_image else "첨부 "
        if file.data is None:
            lines.append((f"{prefix}{file.seq}. {file.name} — {file.problem}", "가져오지 못함"))
            continue
        item = library.items[file.seq]
        label = _format_label(item.kind, item.scanned, item.error)
        lines.append((prefix + listing[file.seq], label))
    return lines


def _format_label(kind: str, scanned: AttachmentText | None, error: str | None) -> str:
    name = KIND_LABELS.get(kind, kind.upper())
    if error is not None or scanned is None:
        return f"못 읽음({name})"
    if kind == "pdf":
        return "PDF 스캔본(Vision)" if scanned.needs_vision else "PDF 글자"
    if kind in IMAGE_KINDS:
        return "이미지(Vision)"
    return name


def _notes(saved: SavedNotice) -> list[str]:
    """사람이 한 번 볼 것: 이름이 달라 짝지은 첨부, 안 쓰는 파일, 뺀 본문 이미지."""
    notes = [
        f"파일 이름이 달라 짝지음: {file.file_name} → 첨부 {file.seq}. {file.name}"
        for file in saved.renamed
    ]
    notes += [
        f"이름이 같은 첨부가 여럿이라 내려받은 순서대로 짝지음: {name}. 맞는지 확인하세요"
        for name in saved.same_name_links
    ]
    notes += [
        f"첨부 {', '.join(map(str, seqs))}의 파일 내용이 같아요."
        " 같은 첨부를 두 번 받은 건 아닌지 확인하세요"
        for seqs in saved.identical
    ]
    if saved.duplicate_files:
        notes.append(f"같은 파일을 또 받은 것(안 씀): {', '.join(saved.duplicate_files)}")
    if saved.extra_files:
        notes.append(f"폴더에만 있는 파일(안 씀): {', '.join(saved.extra_files)}")
    notes += [f"본문 이미지 뺌: {name}" for name in saved.skipped_images]
    if saved.nested_folders:
        notes.append(f"안쪽 폴더의 페이지는 쓰지 않음: {', '.join(saved.nested_folders)}")
    return notes

async def import_folders(

    options: ImportOptions,
    *,
    model: str | None = None,

) -> int:

    if not settings.database_url:

    cost = Decimal(0)

    except psycopg.OperationalError as exc:
        print(f"DB에 연결하지 못했어요: {exc}")
        return 1

        departments = await load_departments(conn)
        try:
            agent = make_agent(settings, departments=departments, fetch_page=None, model=model)

            print(str(exc))
            return 1

        print(f"모델 {agent.model} · 학과 {len(departments)}개 · 공지 폴더 {len(folders)}개")
        if not departments:
            print("학과 목록이 비어 있어요. 학과 조건은 모두 원문 확인 필요가 돼요")
        problems = _duplicates(folders)
        for no, folder in enumerate(folders, 1):
            print(f"[{no}/{len(folders)}] {folder.name}")
            saved = _load(folder, problems)
            if isinstance(saved, str):
                print(f"    오류: {saved}")
                counts["error"] += 1
                continue
            print(f"    {saved.page.title} (글 {saved.page.entry_id})")
            for note in _notes(saved):
                print(f"    {note}")
            try:
                result = await import_notice(
                    conn, using(log_conn), source_id, saved, agent, options
                )
            except Exception as exc:  # LLM·DB 오류. 공고에는 쓰지 않았고 다음에 다시 돈다
                print(f"    오류: 요건 추출 실패 — {type(exc).__name__}: {str(exc)[:200]}")
                counts["error"] += 1
                continue
            counts[result.status] += 1
            counts["unjudged"] += result.unjudged
            cost += result.cost_usd
            for line in _result_lines(result):
                print(f"    {line}")
        await mark_collected(conn, source_id)
    finally:
        await conn.close()
        await log_conn.close()
    names = {"new": "새 공고", "updated": "다시 추출", "unchanged": "그대로", "skipped": "건너뜀"}
    summary = " · ".join(f"{label} {counts[key]}" for key, label in names.items())
    print(
        f"\n{summary} · 오류 {counts['error']} · 판정 안 됨 {counts['unjudged']}"
        f" · 추정 비용 ${cost:.4f}(목록 단가)"
    )
    return 1 if counts["error"] or counts["unjudged"] else 0


def _result_lines(result: ImportResult) -> list[str]:
    extraction = result.extraction
    if extraction is None:
        word = {"unchanged": "그대로", "skipped": "건너뜀", "error": "오류"}[result.status]
        return [f"{word}: {result.message}"]
    if result.status == "new":
        head = "새 공고"
    else:
        head = f"수정된 글 → 다시 추출(요건 버전 {result.version})"
    if not extraction.succeeded:
        return [
            f"{head} · 추출 실패(제출 없음) → 판정하지 않음 · ${result.cost_usd:.4f}",
            result.message,
        ]
    deadline = (
        extraction.deadline_at.astimezone(KST).strftime("%Y-%m-%d %H:%M")
        if extraction.deadline_at
        else "없음"
    )
    lines = [
        f"{head} · 요건 {len(extraction.requirements)}개 · 서류 {len(extraction.documents)}개"
        f" · 마감 {deadline} · 신뢰도 {extraction.confidence or '-'} · ${result.cost_usd:.4f}"
    ]
    if extraction.needs_review:
        lines.append("원문 확인 필요: " + ("; ".join(extraction.review_reasons) or "모델 판단"))
    return lines

def _duplicates(folders: Sequence[Path]) -> dict[Path, str]:
    """같은 글을 두 폴더에 저장했으면 둘 다 멈춘다. 그대로 두면 실행마다 번갈아 다시 추출하거나
    옛 저장본으로 되돌린다. 페이지가 여럿이라 읽지 못하는 폴더의 글 번호도 센다."""
    found: dict[str, list[Path]] = {}
    for folder in folders:
        try:
            ids = entry_ids(folder)
        except Exception:  # 폴더를 읽을 때 이유를 다시 알린다
            continue
        for entry_id in ids:
            found.setdefault(entry_id, []).append(folder)
    problems: dict[Path, str] = {}
    for entry_id, group in found.items():
        if len(group) > 1:
            names = ", ".join(folder.name for folder in group)
            for folder in group:
                problems[folder] = (
                    f"같은 글({entry_id})이 폴더 {len(group)}개에 있어요({names})."
                    " 최근에 저장한 폴더 하나만 남기세요"
                )
    return problems

        return f"폴더를 읽다가 오류가 났어요({type(exc).__name__}: {str(exc)[:200]})"
```

```python
    conn: AsyncConnection,
    log_connect: LogConnect,
    source_id: str,
    saved: SavedNotice,
    agent: ExtractionAgent,
    options: ImportOptions | None = None,
) -> ImportResult:
    """conn은 데이터, log_connect는 실행 로그 연결 공급자(autocommit)다. LLM 호출 오류는 그대로 올라간다."""
    options = options or ImportOptions()
    page = saved.page
    existing = await _existing(conn, source_id, page.entry_id)
    if page.scope == "seoul":
        if existing and await _hide(conn, str(existing["id"])):
            return ImportResult(
                "skipped",
                "서울캠퍼스 공지로 바뀌어 피드에서 숨김(status = hidden)",
                opportunity_id=str(existing["id"]),
            )
        return ImportResult("skipped", "서울캠퍼스 공지라 건너뜀")
    if page.scope == "unmarked" and existing is None and not options.include_unmarked:
        return ImportResult(
            "skipped",
            f"캠퍼스 표시가 없어 건너뜀(캠퍼스 게시: {page.campus or '없음'})."
            " ERICA 학생 공지가 맞으면 --include-unmarked로 가져온다",
        )

    notice = saved.notice_input()
    digest = content_hash(notice.body, notice.attachments)
    if existing and existing["content_hash"] == digest and not options.force:
        opportunity_id = str(existing["id"])
        async with conn.transaction():
            await _refresh_listing(conn, opportunity_id, saved)
            await _refresh_attachments(conn, opportunity_id, saved)
            restored = await _restore_status(conn, opportunity_id, changed=False)
        unjudged = existing["extraction_run_id"] is None
        message = "내용이 같아 다시 추출하지 않음"
        message += f". {RESTORED}" if restored else ""
        message += f". {RETRY_HINT}" if unjudged else ""
        return ImportResult("unchanged", message, opportunity_id=opportunity_id, unjudged=unjudged)
    if saved.missing and not options.allow_missing:
        names = ", ".join(f"{file.seq}. {file.name}({file.problem})" for file in saved.missing)
        return ImportResult(
            "error",
            f"가져오지 못한 파일: {names}. 고치고 다시 실행하거나, 없이 가져오려면 --allow-missing",
        )

        extraction = _note_missing(extraction, saved)

        ids = await _replace_attachments(conn, opportunity_id, saved, extraction)
        extraction = replace(
            extraction,
            attachments=tuple(
                replace(outcome, id=ids[outcome.seq]) for outcome in extraction.attachments
            ),
        )
        version = await save_extraction(
            conn, opportunity_id, extraction, run_id=run.id, content_hash=digest
        )
        restored = await _restore_status(conn, opportunity_id, changed=True)
    notes = [RESTORED] if restored else []
    notes += [] if extraction.succeeded else [RETRY_HINT]
    return ImportResult(
        "new" if existing is None else "updated",
        ". ".join(notes),
        opportunity_id=opportunity_id,
        run_id=run.id,
        version=version,
        extraction=extraction,
        cost_usd=run.cost_usd,
        unjudged=not extraction.succeeded,
    )

async def _refresh_attachments(
    conn: AsyncConnection, opportunity_id: str, saved: SavedNotice
) -> None:
    """내용(파일)이 같아도 학교가 같은 파일을 새 이름·주소로 다시 올렸을 수 있다. 이름·주소만 맞춘다.
    두 행이 주소를 맞바꾸면 (opportunity_id, source_url) 유일 제약에 걸리므로 바뀌는 행의 주소를 먼저 비운다."""
    rows = await (
        await conn.execute(
            "select seq, file_name, source_url from opportunity_attachments"
            " where opportunity_id = %s",
            (opportunity_id,),
        )
    ).fetchall()
    current = {row["seq"]: (row["file_name"], row["source_url"]) for row in rows}
    changed = [
        file
        for file in saved.files
        if file.seq in current and current[file.seq] != (file.name, file.source_url)
    ]
    if not changed:
        return
    await conn.execute(
        "update opportunity_attachments set source_url = null"
        " where opportunity_id = %s and seq = any(%s)",
        (opportunity_id, [file.seq for file in changed]),
    )
    for file in changed:
        await conn.execute(
            "update opportunity_attachments set file_name = %s, source_url = %s"
            " where opportunity_id = %s and seq = %s",
            (file.name, file.source_url, opportunity_id, file.seq),

            " organizer, original_url, raw_text) values (%(source_id)s, 'school_notice',"
            " %(entry_id)s, %(category)s::opportunity_category, %(title)s, %(organizer)s,"
            " %(url)s, %(raw_text)s)"
            " on conflict (source_id, external_id) do update set category = excluded.category,"
            " title = excluded.title, organizer = excluded.organizer,"
            " original_url = excluded.original_url, raw_text = excluded.raw_text"
            " returning id",
            {
                **_listing(saved),
                "source_id": source_id,
                "entry_id": saved.page.entry_id,
                "raw_text": body,
            },
        )
    ).fetchone()
    return str(row["id"])


async def _replace_attachments(
    conn: AsyncConnection, opportunity_id: str, saved: SavedNotice, extraction: ExtractionResult
) -> dict[int, str]:
    """첨부 행을 페이지 순서대로 다시 만든다. 첨부 번호 → 행 id."""
    await conn.execute(
        "delete from opportunity_attachments where opportunity_id = %s", (opportunity_id,)
    )
    outcomes = {outcome.seq: outcome for outcome in extraction.attachments}
    ids: dict[int, str] = {}
    for file in saved.files:
        if file.data is None:  # 가져오지 못한 파일: 실패 사유를 남겨 원문 확인 필요가 보이게 한다
            values = (
                opportunity_id, file.seq, file.name, None, file.source_url, None,
                f"파일을 가져오지 못해 읽지 않았어요({file.problem})", True,
            )  # fmt: skip
        else:
            outcome = outcomes.get(file.seq)
            values = (
                opportunity_id, file.seq, file.name, outcome.mime_type if outcome else None,
                file.source_url, hashlib.sha256(file.data).hexdigest(), None, False,
            )  # fmt: skip
        row = await (
            await conn.execute(
                "insert into opportunity_attachments (opportunity_id, seq, file_name, mime_type,"
                " source_url, sha256, extract_error, extracted_at) values"
                " (%s, %s, %s, %s, %s, %s, %s, case when %s then now() end) returning id",
                values,
            )
        ).fetchone()
        ids[file.seq] = str(row["id"])
    return ids


def _note_missing(extraction: ExtractionResult, saved: SavedNotice) -> ExtractionResult:
    # 학생에게 보이는 문구다(공고 상세 review_reasons). 본문 이미지는 상세의 첨부 목록에 없어서
    # 번호 대신 파일 이름으로 쓴다
    names = ", ".join(file.name for file in saved.missing)
    return replace(
        extraction,
        needs_review=True,
        review_reasons=(*extraction.review_reasons, f"가져오지 못해 읽지 않은 첨부: {names}"),
    )
```

```python
def create_pools(database_url: str) -> tuple[AsyncConnectionPool, AsyncConnectionPool]:

    )

    if pool is None:
        raise ApiError(503, "DB_UNAVAILABLE", "데이터베이스에 연결할 수 없어요.")
    async with pool.connection() as conn:  # 블록이 끝나면 커밋, 예외면 롤백
        yield conn


async def get_conn(request: Request) -> AsyncIterator[AsyncConnection]:
    async for conn in _connection(request.app.state.pool):
        yield conn


def get_log_connect(request: Request) -> LogConnect:
    """agent_run()에 넘기는 autocommit 연결 공급자.

    연결 하나가 아니라 공급자를 준다. 로깅은 한 줄 쓸 때마다 빌렸다 돌려줘서, 모델을 기다리는
    오래 걸리는 실행이 로그 풀(max_size)을 붙잡지 않는다(app/agent_log.py).
    """
    log_pool = getattr(request.app.state, "log_pool", None)
    if log_pool is None:
        raise ApiError(503, "DB_UNAVAILABLE", "데이터베이스에 연결할 수 없어요.")
    return log_pool.connection
```

```python
    if isinstance(value, Decimal):
```

#### `api/app/prepare/service.py`

```python
"""준비하기 실행(PRD 6장 "준비하기 버튼": get_document_guide → build_prep_plan).

POST /opportunities/{id}/prepare(라우터)가 판정·확인을 마치면 begin이 실행 행과 계획 행을 만들고 일을 띄운다.
일(execute)은 요청과 따로 돌며 자기 연결을 연다.

일은 읽기 → 모델 → 쓰기 셋으로 나눈다. 모델 호출은 트랜잭션 밖이라 묻는 동안 DB 연결(요청 풀)과 공고 행
잠금을 쥐지 않는다. 쓰기(서류 안내 + 할 일)는 공고를 잠근 한 트랜잭션이고, 잠근 뒤 서류를 다시 읽어 그새
다른 학생이 넣거나 채운 값을 그대로 쓴다. 실패하면 아무것도 남지 않고 계획 행을 지워 다시 누를 수 있다.
실행 기록(agent_runs·tool_calls·llm_calls)은 실패해도 남고, running으로 남지 않게 닫는다.
로그 연결은 한 줄 쓸 때마다 빌렸다 돌려준다(agent_log). 모델을 기다리는 동안 쥐고 있지 않는다.
"""

import logging
from collections.abc import Callable
from contextlib import AbstractAsyncContextManager, AsyncExitStack
from dataclasses import dataclass, field
from datetime import date, datetime
from typing import Annotated

from fastapi import Depends, Request
from psycopg import AsyncConnection

from app.agent_log import AgentRun, ToolCall, abandon_run, agent_run, start_run
from app.background import RunnerDep, TaskRunner
from app.config import Settings, get_settings
from app.errors import ApiError
from app.llm import LlmClient
from app.llm.deps import LlmDep
from app.prepare.documents import Guide, lookup
from app.prepare.guide import GuideDraft, GuideError, ask_guides
from app.prepare.plan import PlanDocument, build_tasks
from app.repositories.opportunities import OpportunityRepository
from app.repositories.prepare import DocumentRow, OpportunityText, PlanState, PrepareRepository
from app.schemas.opportunities import OpportunityDetail
from app.youth.convert import raw_section

logger = logging.getLogger(__name__)
Connect = Callable[[], AbstractAsyncContextManager[AsyncConnection]]
GUIDE_FAILED = "필요 서류를 확인하지 못했어요. 잠시 후 다시 눌러 주세요."
STOPPED = "준비 일정을 만들다 멈췄어요. 다시 눌러 주세요."


@dataclass(frozen=True)
class PrepareJob:
    run_id: str
    plan_id: str
    user_id: str
    opportunity_id: str
    deadline: date  # 마감일(KST 날짜). 상시 공고는 학생이 고른 목표일
    today: date  # 요청한 날(KST). 이보다 앞선 할 일은 이날로 둔다


@dataclass(frozen=True)
class Asked:
    """모델에 물어 온 것. 쓰기 트랜잭션을 열기 전에 다 받아 둔다."""

    listed: list[GuideDraft] | None = None  # 청년정책 '제출 서류' 글에서 만든 서류 목록
    guides: dict[str, GuideDraft] = field(default_factory=dict)  # 사전에 없던 서류의 이름 → 안내


@dataclass(frozen=True)
class Started:
    run_id: str | None
    status: str  # running | succeeded
    created: bool  # 이번 요청이 시작했다(202). 아니면 그새 다른 요청이 시작한 계획(200)


class PrepareService:
    def __init__(
        self,
        *,
        transaction: Connect,
        log_connection: Connect,
        llm: LlmClient | None,
        model: str,
        runner: TaskRunner,
    ) -> None:
        self.transaction = transaction  # 블록이 끝나면 커밋하는 연결(요청 트랜잭션과 따로)
        self.log_connection = log_connection  # autocommit 연결 공급자(agent_run이 쓸 때마다 빌린다)
        self.llm = llm
        self.model = model
        self.runner = runner

    async def judge(
        self, user_id: str, opportunity_id: str, *, now: datetime, display_name: str | None
    ) -> OpportunityDetail | None:
        """상세와 같이 지금 프로필로 다시 판정한다(프로필 잠금 포함). 보이지 않는 공고면 None.

        요청 트랜잭션을 쓰지 않고 바로 커밋한다. 뒤에서 도는 일과 잠금을 나눠 갖지 않게 한다.
        """
        async with self.transaction() as conn:
            return await OpportunityRepository(conn).detail(
                user_id, opportunity_id, now=now, display_name=display_name
            )

    async def plan(self, user_id: str, opportunity_id: str) -> PlanState | None:
        async with self.transaction() as conn:
            return await PrepareRepository(conn).plan(user_id, opportunity_id)

    async def reset(self, plan: PlanState) -> None:
        """멈췄거나 실패한 계획을 지운다. 실행 중으로 남은 실행은 실패로 닫는다."""
        async with self.transaction() as conn:
            await PrepareRepository(conn).drop_plan(plan.id)
        if plan.run_id:
            await self._close(plan.run_id)

    async def begin(
        self, user_id: str, opportunity_id: str, *, deadline: date, today: date
    ) -> Started:
        """실행 행과 계획 행을 만들고 일을 띄운다. 그새 다른 요청이 계획을 만들었으면 그 계획을 준다."""
        run_id = await start_run(self.log_connection, trigger="prepare", user_id=user_id)
        try:
            async with self.transaction() as conn:
                plan_id = await PrepareRepository(conn).create_plan(user_id, opportunity_id, run_id)
        except Exception:  # 계획 행을 만들지 못했다. 실행이 running으로 남지 않게 닫는다
            await self._close(run_id)
            raise
        if plan_id is None:
            await self._close(run_id, "같은 공고를 이미 준비하고 있어요.")
            existing = await self.plan(user_id, opportunity_id)
            if existing is None:  # 그 계획도 그새 실패해 지워졌다. 다시 누르면 된다
                raise ApiError(500, "INTERNAL_ERROR", STOPPED)
            status = "succeeded" if existing.done else "running"
            return Started(existing.run_id, status, created=False)
        job = PrepareJob(run_id, plan_id, user_id, opportunity_id, deadline, today)
        work = self.execute(job)
        try:
            await self.runner.start(work)
        except Exception:  # 일을 띄우지 못했다. 계획을 지워 다시 누를 수 있게 한다
            work.close()
            await self._give_up(job)
            raise
        return Started(run_id, "running", created=True)

    async def execute(self, job: PrepareJob) -> None:
        """서류 안내 → 역산 일정. 실패하면 계획을 지운다(실행은 실패로 남는다)."""
        try:
            async with agent_run(
                self.log_connection, trigger="prepare", user_id=job.user_id, run_id=job.run_id
            ) as run:
                await self._prepare(run, job)
        except Exception:
            logger.exception("준비하기 실행 실패: run %s", job.run_id)
            await self._give_up(job)

    async def _give_up(self, job: PrepareJob) -> None:
        """계획을 지우고 실행을 닫는다. 로그 연결을 얻지 못해 agent_run 밖에서 끝난 경우까지 닫는다."""
        try:
            async with self.transaction() as conn:
                await PrepareRepository(conn).drop_plan(job.plan_id)
        except Exception:
            logger.exception("실패한 준비 계획을 지우지 못했어요: plan %s", job.plan_id)
        await self._close(job.run_id)

    async def _close(self, run_id: str, message: str = STOPPED) -> None:
        """아직 running인 실행을 실패로 닫는다(이미 닫혔으면 그대로 둔다).

        닫지 못해도 다음 누름이 멈춘 실행으로 보고 지운다(repositories/prepare.py의 STALE_AFTER).
        """
        try:
            await abandon_run(self.log_connection, run_id, ApiError(500, "INTERNAL_ERROR", message))
        except Exception:
            logger.exception("준비하기 실행을 닫지 못했어요: run %s", run_id)

    async def _prepare(self, run: AgentRun, job: PrepareJob) -> None:
        """읽기 → 모델 → 쓰기. 쓰기 트랜잭션은 모델 호출이 끝난 뒤에 열어 두 단계가 함께 커밋된다."""
        async with AsyncExitStack() as stack:
            async with run.tool(
                "get_document_guide", {"opportunity_id": job.opportunity_id}
            ) as call:
                asked = await self._ask_guides(run, call, job)  # 트랜잭션 밖(모델)
                conn = await stack.enter_async_context(self.transaction())
                repo = PrepareRepository(conn)
                if not await repo.lock_plan(job.plan_id):
                    # 묻는 사이에 다음 누름이 멈춘 실행으로 보고 계획을 지웠다. 그 요청에 넘긴다
                    raise ApiError(500, "INTERNAL_ERROR", STOPPED)
                documents, counts = await self._save_guides(repo, job, asked)
                call.output = {"documents": len(documents), **counts}
            plan_input = {
                "opportunity_id": job.opportunity_id,
                "deadline": job.deadline.isoformat(),
                "documents": len(documents),
            }
            async with run.tool("build_prep_plan", plan_input) as call:
                tasks = build_tasks(
                    [PlanDocument(doc.id, doc.name, doc.lead_days) for doc in documents],
                    job.deadline,
                    job.today,
                )
                await repo.add_tasks(job.user_id, job.plan_id, tasks)
                call.output = {"tasks": len(tasks)}

    async def _ask_guides(self, run: AgentRun, call: ToolCall, job: PrepareJob) -> Asked:
        """모델에 물을 것만 묻는다. 공고 행을 잠그지 않고, 묻는 동안 DB 연결도 쥐지 않는다."""
        async with self.transaction() as conn:
            repo = PrepareRepository(conn)
            opportunity = await repo.opportunity(job.opportunity_id)
            if opportunity is None:  # 그새 공고가 지워졌다
                raise ApiError(404, "NOT_FOUND", "공고를 찾을 수 없어요.")
            documents = await repo.documents(job.opportunity_id)
        if not documents and opportunity.source_type == "youth_policy":
            text = raw_section(opportunity.raw_text, "제출 서류")
            if text:  # 청년정책은 서류 행이 없다. 제출 서류 글에서 목록과 안내를 함께 만든다
                drafts = await self._ask(run, call, opportunity, text=text)
                # 묻는 사이에 서류 행이 생겼으면 목록은 버리고 안내만 이름으로 맞춰 쓴다
                return Asked(listed=drafts, guides={draft.name: draft for draft in drafts})
        unknown = [
            document.name
            for document in documents
            if not document.guided and lookup(document.name) is None
        ]
        if not unknown:  # 사전으로 다 되거나 이미 채워진 공고는 모델을 부르지 않는다
            return Asked()
        drafts = await self._ask(run, call, opportunity, names=unknown)
        return Asked(guides={draft.name: draft for draft in drafts})

    async def _save_guides(
        self, repo: PrepareRepository, job: PrepareJob, asked: Asked
    ) -> tuple[list[DocumentRow], dict[str, int]]:
        """공고를 잠그고 서류를 다시 읽어 빈 칸만 채운다. 채운 값은 공고에 남아 다른 학생도 쓴다."""
        if await repo.lock_opportunity(job.opportunity_id) is None:  # 그새 공고가 지워졌다
            raise ApiError(404, "NOT_FOUND", "공고를 찾을 수 없어요.")
        documents = await repo.documents(job.opportunity_id)
        counts = {"dictionary": 0, "model": 0, "unknown": 0, "listed": 0}
        if asked.listed is not None and not documents:  # 그새 다른 학생이 넣었으면 그 서류를 쓴다
            drafts = [_with_dictionary(draft) for draft in asked.listed]
            await repo.add_documents(job.opportunity_id, drafts)
            counts["listed"] = len(drafts)
            documents = await repo.documents(job.opportunity_id)
        guides: list[tuple[str, Guide]] = []
        for document in documents:
            if document.guided:  # 이미 채워져 있다(다른 학생이 준비한 공고)
                continue
            if guide := lookup(document.name):
                guides.append((document.id, guide))
                counts["dictionary"] += 1
            elif draft := asked.guides.get(document.name):
                guides.append((document.id, _guide(draft)))
                counts["model"] += 1
            else:  # 모델도 답하지 않은 서류. 소요 일수는 계획에서 하루로 둔다
                counts["unknown"] += 1
        await repo.save_guides(guides)
        return await repo.documents(job.opportunity_id), counts

    async def _ask(
        self,
        run: AgentRun,
        call: ToolCall,
        opportunity: OpportunityText,
        *,
        names: list[str] | None = None,
        text: str | None = None,
    ) -> list[GuideDraft]:
        if self.llm is None:
            raise ApiError(502, "LLM_ERROR", GUIDE_FAILED)
        try:
            drafts, reply = await ask_guides(
                self.llm,
                model=self.model,
                title=opportunity.title,
                organizer=opportunity.organizer,
                names=names,
                text=text,
            )
        except TimeoutError as exc:
            raise ApiError(502, "LLM_TIMEOUT", GUIDE_FAILED) from exc
        except GuideError as exc:
            raise ApiError(502, "LLM_ERROR", GUIDE_FAILED) from exc
        except Exception as exc:
            logger.warning("서류 안내 모델 호출 실패: %r", exc)
            raise ApiError(502, "LLM_ERROR", GUIDE_FAILED) from exc
        await run.record_llm(
            provider=self.llm.provider,
            model=self.model,
            input_tokens=reply.usage.input_tokens,
            output_tokens=reply.usage.output_tokens,
            latency_ms=reply.latency_ms,
            tool_call=call,
        )
        return drafts


def _guide(draft: GuideDraft) -> Guide:
    return Guide(draft.issuer, draft.how_to, draft.lead_days, draft.effort_minutes)


def _with_dictionary(draft: GuideDraft) -> GuideDraft:
    """사전에 있는 서류는 모델 답 대신 사전 안내를 쓴다(이름과 필수 여부는 모델 답)."""
    if guide := lookup(draft.name):
        return GuideDraft(
            draft.name,
            guide.issuer,
            guide.how_to,
            guide.lead_days,
            guide.effort_minutes,
            draft.is_required,
        )
    return draft


def get_prepare_service(
    request: Request,
    llm: LlmDep,
    runner: RunnerDep,
    settings: Annotated[Settings, Depends(get_settings)],
) -> PrepareService:
    state = request.app.state
    pool, log_pool = getattr(state, "pool", None), getattr(state, "log_pool", None)
    if pool is None or log_pool is None:
        raise ApiError(503, "DB_UNAVAILABLE", "데이터베이스에 연결할 수 없어요.")
    return PrepareService(
        transaction=pool.connection,
        log_connection=log_pool.connection,
        llm=llm,

        runner=runner,
    )


PrepareServiceDep = Annotated[PrepareService, Depends(get_prepare_service)]
```

```python
"""온통청년 정책을 공고로 넣는다 (PRD F-11). 수집 한 번이 실행 하나(agent_runs.trigger_type = batch_crawl)다.

실행의 도구 호출은 둘이다. 넣은 공고의 extraction_run_id가 이 실행이라 공고 상세의 처리 과정에 그대로 보인다.
1. fetch_youth_policies: 온통청년에서 정책 목록을 끝까지 받는다(client.py)
2. build_youth_requirements: 가져올 정책을 고르고 요건을 만들어 저장한다(convert.py)

정책마다 한 트랜잭션이다.
- 새 정책·바뀐 정책(변환 결과의 content_hash가 다름): 공고 행을 넣거나 고치고 요건을 바꾼다(save_extraction).
  요건이 바뀌면 requirements_version이 오르고, 판정은 피드를 읽을 때 다시 계산된다.
- 같은 정책: 고치지 않는다. 조회수처럼 변환에 안 쓰는 칸만 바뀐 정책이다. 넣은 정책의 content_hash·상태를
  먼저 한 번에 읽어, 같고 활성인 정책은 DB에 다시 가지 않는다(원격 DB에서 정책마다 오가면 느리다).
- 가져온 정책이 숨김·만료였으면 활성으로 돌린다(만료는 마감이 지나지 않았을 때만). 상시 정책이 마감 코드로
  바뀌었다가 다시 상시가 되면 변환 결과가 전과 같아서, 같은 정책이어도 돌린다.
- 이미 넣은 정책이 범위를 벗어나면 마감·사업 종료는 만료, 수도권 밖·승인 안 됨은 숨김으로 바꾼다.
- 목록을 빠짐없이 받았을 때만 목록에서 사라진 정책을 숨긴다. 한 번에 활성 공고의 GONE_SHARE(최소 GONE_MIN건)
  넘게 사라졌으면 온통청년 목록이 덜 온 것일 수 있어 숨기지 않고 수만 남긴다(gone_kept). 온통청년에서 정말
  내려간 정책이면 hide_gone(명령의 --hide-gone)으로 숨긴다.
"""

from collections import Counter
from dataclasses import dataclass, field
from datetime import date
from typing import Literal

from psycopg import AsyncConnection

from app.agent_log import LogConnect, agent_run
from app.extraction.models import ExtractionResult
from app.extraction.store import save_extraction
from app.youth import BUILD_TOOL, FETCH_TOOL
from app.youth.client import BASE_URL, FetchResult, YouthPolicyClient
from app.youth.convert import PolicyRecord, Skip, convert
from app.youth.regions import region_table

SOURCE_TYPE = "youth_policy"
SOURCE_NAME = "온통청년"
SCOPE = "마감 전·상시, 전국+수도권"
GONE_MIN = 20
GONE_SHARE = 0.2

Saved = Literal["new", "updated", "unchanged"]


@dataclass
class CollectResult:
    run_id: str
    fetch: FetchResult
    counts: Counter[str] = field(default_factory=Counter)  # new·updated·unchanged·expired·hidden
    skips: Counter[str] = field(default_factory=Counter)  # 가져오지 않은 이유별 정책 수
    gone_kept: int = 0  # 목록에서 너무 많이 사라져 숨기지 않은 정책 수


async def ensure_source(conn: AsyncConnection) -> str:
    """sources의 온통청년 행. 없으면 만든다."""
    row = await (
        await conn.execute(
            "select id from sources where type = %s and base_url = %s order by id limit 1",
            (SOURCE_TYPE, BASE_URL),
        )
    ).fetchone()
    if row is None:
        row = await (
            await conn.execute(
                "insert into sources (type, name, base_url) values (%s, %s, %s) returning id",
                (SOURCE_TYPE, SOURCE_NAME, BASE_URL),
            )
        ).fetchone()
    return str(row["id"])


async def collect(
    conn: AsyncConnection,
    log_connect: LogConnect,
    client: YouthPolicyClient,
    *,
    today: date,
    scenario: str | None = None,
    hide_gone: bool = False,
) -> CollectResult:
    """conn은 데이터(autocommit, 정책마다 트랜잭션), log_connect는 실행 로그 연결 공급자다.
    today는 KST 날짜다.
    온통청년 오류(YouthApiError)는 실행을 실패로 남기고 그대로 올라간다."""
    source_id = await ensure_source(conn)
    table = region_table()
    async with agent_run(log_connect, trigger="batch_crawl", scenario_code=scenario) as run:
        async with run.tool(FETCH_TOOL, {"page_size": 100, "scope": SCOPE}) as call:
            fetched = await client.fetch_all()
            call.output = fetched.summary()
        result = CollectResult(run_id=run.id, fetch=fetched)
        tool_input = {"policies": len(fetched.policies), "hide_gone": hide_gone}
        async with run.tool(BUILD_TOOL, tool_input) as call:
            known = await _known(conn, source_id)
            skipped: dict[str, Skip] = {}
            for policy in fetched.policies:
                converted = convert(policy, today, table)
                if isinstance(converted, Skip):
                    skipped[str(policy["plcyNo"])] = converted
                    result.skips[converted.reason] += 1
                elif known.get(converted.external_id) == (converted.content_hash, "active"):
                    result.counts["unchanged"] += 1  # 같고 활성인 공고는 DB에 다시 가지 않는다
                else:
                    result.counts[await save_policy(conn, source_id, converted, run.id)] += 1
            expired, hidden, result.gone_kept = await retire(
                conn,
                source_id,
                skipped,
                {str(policy["plcyNo"]) for policy in fetched.policies},
                known,
                complete=fetched.complete,
                hide_gone=hide_gone,
            )
            result.counts["expired"] += expired
            result.counts["hidden"] += hidden
            await conn.execute(
                "update sources set last_collected_at = now() where id = %s", (source_id,)
            )
            call.output = {
                "saved": dict(result.counts),
                "skipped": dict(result.skips),
                "gone_kept": result.gone_kept,
            }
    return result


async def _known(conn: AsyncConnection, source_id: str) -> dict[str, tuple[str | None, str]]:
    """이미 넣은 정책: 번호 → (content_hash, 상태). 정책마다 DB를 오가지 않게 한 번에 읽는다."""
    rows = await (
        await conn.execute(
            "select external_id, content_hash, status::text from opportunities"
            " where source_id = %s",
            (source_id,),
        )
    ).fetchall()
    return {row["external_id"]: (row["content_hash"], row["status"]) for row in rows}


async def save_policy(
    conn: AsyncConnection, source_id: str, record: PolicyRecord, run_id: str
) -> Saved:
    """정책 한 건을 한 트랜잭션으로 넣는다. 제목·기관·주소도 변환 결과(content_hash)에 들어 있다."""
    async with conn.transaction():
        existing = await (
            await conn.execute(
                "select id, content_hash from opportunities"
                " where source_id = %s and external_id = %s for update",
                (source_id, record.external_id),
            )
        ).fetchone()
        saved: Saved = "new" if existing is None else "updated"
        if existing is not None and existing["content_hash"] == record.content_hash:
            opportunity_id, saved = str(existing["id"]), "unchanged"
        else:
            opportunity_id = await _upsert(conn, source_id, record)
            await save_extraction(
                conn,
                opportunity_id,
                _as_extraction(record),
                run_id=run_id,
                content_hash=record.content_hash,
            )
        await conn.execute(_RESTORE, (opportunity_id,))
    return saved


async def _upsert(conn: AsyncConnection, source_id: str, record: PolicyRecord) -> str:
    row = await (
        await conn.execute(
            "insert into opportunities (source_id, source_type, external_id, category, title,"
            " organizer, original_url, raw_text) values (%(source_id)s, 'youth_policy',"
            " %(external_id)s, 'youth_policy', %(title)s, %(organizer)s, %(url)s,"
            " %(raw_text)s) on conflict (source_id, external_id) do update set"
            " title = excluded.title, organizer = excluded.organizer,"
            " original_url = excluded.original_url, raw_text = excluded.raw_text"
            " returning id",
            {
                "source_id": source_id,
                "external_id": record.external_id,
                "title": record.title,
                "organizer": record.organizer,
                "url": record.original_url,
                "raw_text": record.raw_text,
            },
        )
    ).fetchone()
    return str(row["id"])


async def retire(
    conn: AsyncConnection,
    source_id: str,
    skipped: dict[str, Skip],
    fetched: set[str],
    known: dict[str, tuple[str | None, str]],
    *,
    complete: bool,
    hide_gone: bool = False,
) -> tuple[int, int, int]:
    """범위를 벗어난 활성 공고를 만료·숨김으로 바꾼다. (만료 수, 숨김 수, 숨기지 않은 사라진 정책 수).

    known은 이번 수집 전에 읽은 넣은 정책(_known)이다. 목록에서 사라진 정책은 빠짐없이 받았을 때만 숨기고,
    한 번에 너무 많이 사라졌으면 hide_gone일 때만 숨긴다."""
    closed = [number for number, skip in skipped.items() if skip.expired]
    out = [number for number, skip in skipped.items() if not skip.expired]
    expired = (await conn.execute(_EXPIRE, (source_id, closed))).rowcount
    hidden = (await conn.execute(_HIDE, (source_id, out))).rowcount
    if not complete:
        return expired, hidden, 0
    active = {number for number, (_, status) in known.items() if status == "active"}
    gone = sorted(active - fetched)
    if not hide_gone and len(gone) > max(GONE_MIN, len(active) * GONE_SHARE):
        return expired, hidden, len(gone)
    hidden += (await conn.execute(_HIDE, (source_id, gone))).rowcount
    return expired, hidden, 0


_RESTORE = (
    "update opportunities set status = 'active' where id = %s and (status = 'hidden'"
    " or (status = 'expired' and (deadline_at is null or deadline_at > now())))"
)
_ACTIVE_OF_SOURCE = "where source_id = %s and status = 'active'"
_EXPIRE = (
    f"update opportunities set status = 'expired' {_ACTIVE_OF_SOURCE} and external_id = any(%s)"

_HIDE = f"update opportunities set status = 'hidden' {_ACTIVE_OF_SOURCE} and external_id = any(%s)"
```

```python
class RecordingConnection:
    """빌린 횟수까지 세는 가짜 로그 연결. connect()가 공급자다."""

    autocommit = True

    def __init__(self) -> None:
        self.calls: list[tuple[str, Any]] = []
        self.borrows = 0

    async def execute(self, query: str, params: Any = None) -> None:
        self.calls.append((query, params))

    def connect(self) -> Any:
        connection = self

        @asynccontextmanager
        async def borrow() -> AsyncIterator["RecordingConnection"]:
            connection.borrows += 1
            yield connection

        return borrow()
```

```python
async def test_a_long_run_does_not_hold_a_log_connection() -> None:
    """모델을 기다리는 동안 로그 연결을 쥐지 않는다. 쥐면 풀만큼만 동시에 돌 수 있다.

    로그 풀을 1칸으로 만들고, 실행 한가운데(모델을 부르는 자리)에서 다른 쪽이 연결을 빌려 본다.
    실행이 연결을 들고 있으면 여기서 PoolTimeout이 난다.
    """
    pool = AsyncConnectionPool(
        DATABASE_URL,
        min_size=1,
        max_size=1,
        open=False,
        kwargs={"row_factory": dict_row, "autocommit": True, "prepare_threshold": None},
    )
    await pool.open()
    run_ids: list[str] = []
    try:
        async with agent_run(pool.connection, trigger="eval", scenario_code="TEST-POOL") as run:
            run_ids.append(run.id)
            async with run.tool("get_document_guide", {"opportunity_id": "x"}) as call:
                # 모델에 묻는 중이라고 치고, 그 사이 다른 요청이 로그 연결을 빌린다
                async with pool.connection(timeout=2) as other:
                    assert (await (await other.execute("select 1 as ok")).fetchone())["ok"] == 1
                call.output = {"documents": 0}

            (row,) = await (
                await conn.execute(
                    "select status::text as status from agent_runs where id = %s", (run_ids[0],)
```

```python
async def test_detail_keeps_the_extraction_order() -> None:

            " returning id",

        assert [c.condition_count for c in detail.eligibility.clauses] == [5, 2]
    finally:
        await conn.rollback()
        await conn.close()


async def test_detail_shows_review_reasons() -> None:
    """원문 확인 필요 이유(S1-6b): 추출할 때마다 새 값으로 바뀌고, 실패한 추출도 이유가 보인다.

    저장은 받은 그대로 하고, 상세는 모델이 쓴 글을 한 줄로 정리해 보인다.
    """
    conn = await connect()
    try:
        me = await make_user(conn)
        opportunity = await one(
            conn,
            "insert into opportunities (source_type, title) values ('school_notice', '이유 장학')"
            " returning id, now() as now",
        )
        run = await one(
            conn,
            "insert into agent_runs (trigger_type, status) values ('batch_crawl', 'succeeded')"
            " returning id",
        )
        assert opportunity is not None and run is not None
        opportunity_id, now, run_id = str(opportunity["id"]), opportunity["now"], str(run["id"])
        repo = OpportunityRepository(conn)

        async def save(result: ExtractionResult, digest: str) -> Any:
            await save_extraction(conn, opportunity_id, result, run_id=run_id, content_hash=digest)
            stored = await one(
                conn, "select review_reasons from opportunities where id = %s", opportunity_id
            )
            detail = await repo.detail(me, opportunity_id, now=now)
            assert stored is not None and detail is not None
            return stored["review_reasons"], detail

        # 새 공고는 이유가 없다(빈 배열)
        fresh = await repo.detail(me, opportunity_id, now=now)
        assert fresh is not None and (fresh.needs_review, fresh.review_reasons) == (False, [])

        raw = (
            "읽지 못한 첨부가 있음",
            "AI 메모: 신청서는\nhttps://evil.example/form 에서 받기",
        )
        flagged = ExtractionResult(
            succeeded=True, confidence=Decimal("0.9"), needs_review=True, review_reasons=raw
        )
        stored, detail = await save(flagged, "h1")
        assert stored == list(raw)  # 저장은 그대로
        assert detail.needs_review and detail.review_reasons == [
            "읽지 못한 첨부가 있음",
            "AI 메모: 신청서는 에서 받기",
        ]

        # 다시 추출해서 이유가 없어지면 비운다
        clean = ExtractionResult(succeeded=True, confidence=Decimal("0.9"))
        stored, detail = await save(clean, "h2")
        assert stored == [] and (detail.needs_review, detail.review_reasons) == (False, [])

        # 추출에 실패해도 이유는 남는다. 처리 과정은 비어 있다(실행이 공고에 이어지지 않음)
        failed = ExtractionResult(
            succeeded=False, needs_review=True, review_reasons=("자격 요건을 정리하지 못함",)
        )
        stored, detail = await save(failed, "h3")
        assert stored == ["자격 요건을 정리하지 못함"]
        assert (detail.needs_review, detail.review_reasons) == (True, ["자격 요건을 정리하지 못함"])
        assert detail.eligibility.reason_text == UNJUDGED_SUMMARY
        assert detail.process.extraction == []
    finally:
        await conn.rollback()
        await conn.close()
```

```python
    finally:
        for http in clients:
            await http.aclose()
        await pool.close()
        await log_pool.close()


async def rows(query: str, *params: Any) -> list[dict[str, Any]]:
    conn = await connect()
    try:
        return await (await conn.execute(query, params)).fetchall()
    finally:
        await conn.close()

def _can_lock(opportunity_id: str) -> bool:
    """다른 연결이 공고 행을 바로 잠글 수 있나(준비하기가 잠금을 쥔 채 모델을 부르지 않는지)."""
    with psycopg.connect(DATABASE_URL, autocommit=True) as conn:
        try:
            conn.execute(
                "select 1 from opportunities where id = %s for no key update nowait",
                (opportunity_id,),
            )
        except psycopg.errors.LockNotAvailable:
            return False
        return True

    )

        DATABASE_URL,

        max_size=1,
        open=False,
        kwargs={"row_factory": dict_row, "autocommit": True, "prepare_threshold": None},
    )
    await pool.open()
    await log_pool.open()

    app.dependency_overrides[get_settings] = lambda: settings
    app.dependency_overrides[get_llm] = lambda: llm
    clients: list[httpx.AsyncClient] = []

    def client(user_id: str) -> httpx.AsyncClient:

            transport=httpx.ASGITransport(app=app),
            base_url="http://test",
            headers={"Authorization": f"Bearer {make_token(sub=user_id)}"},
        )
        clients.append(http)
        return http

    try:
        yield client
    finally:
        await runner.drain(timeout=10)
        for http in clients:
            await http.aclose()
        await pool.close()
        await log_pool.close()
```

```python
async def test_age_is_judged_on_the_days_one_can_apply() -> None:
    """청년정책은 마감일 하나로 나이를 따지지 않는다(판정엔진_규칙.md 8장). 신청할 수 있는 첫날(오늘, 신청 기간
    전이면 시작일)과 마감일 중 맞는 날이 있으면 그 결과다. 끝난 정책은 마감일이다. 날이 바뀌면 다시 판정한다."""
    conn, log_conn = await connect(), await connect(autocommit=True)
    try:
        await skip_if_other_youth_policies(conn, PREFIX)
        older = await make_user(conn, birth_date="1992-01-01")  # 오늘 만 34세
        freshman = await make_user(conn, birth_date="2007-11-15")  # 오늘 만 18세, 11/15에 19세
        late = await make_user(conn, birth_date="1991-10-01")  # 9월에 만 34세, 10/1부터 35세
        teen = await make_user(conn, birth_date="2007-10-05")  # 9월에 만 18세, 10/5부터 19세
        ages = {"sprtTrgtMinAge": "19", "sprtTrgtMaxAge": "34", "aplyPrdSeCd": "0057001"}
        await run(
            conn,
            log_conn,
            item("E", **ages, aplyYmd="20260901 ~ 20260930"),
            today=date(2026, 9, 15),
        )
        await run(
            conn,
            log_conn,
            item("L", **ages, aplyYmd="20260101 ~ 20281231"),  # 몇 년짜리 신청 기간
            item("F", **ages, aplyYmd="20270101 ~ 20270131"),  # 아직 시작 전
            item("Y", **ages, aplyYmd="20261001 ~ 20261117"),  # 마감 전에 19세가 된다
            item("E", **ages, aplyYmd="20260901 ~ 20260930"),  # 끝남 → 만료
        )
        saved = await rows(conn)
        repo = OpportunityRepository(conn)

        async def judged(user: str, name: str) -> tuple[date | None, str]:
            found = (await repo.detail(user, str(saved[name]["id"]), now=NOW)).eligibility
            return found.basis_date, found.display_status

        # 오늘 만 34세라 지원 가능. 마감일(2028-12-31)로만 따지면 만 36세라 지원 어려움이 된다
        assert await judged(older, "L") == (TODAY, "eligible")
        assert await judged(older, "F") == (date(2027, 1, 1), "ineligible")  # 시작일에 만 35세
        # 오늘은 만 18세지만 마감일엔 19세: 지원 가능, 기준일은 마감일
        assert await judged(freshman, "Y") == (date(2026, 11, 17), "eligible")
        # 끝난 정책은 마감일(9/30, 만 34세)로 본다. 오늘(만 35세)로 다시 따지지 않는다
        assert saved["E"]["status"] == "expired"
        assert await judged(late, "E") == (date(2026, 9, 30), "eligible")
        assert await judged(teen, "E") == (date(2026, 9, 30), "ineligible")  # 오늘 19세여도

        # 상세에서 판정한 L·F는 그대로, 처음 보는 Y만 판정한다. 같은 날 다시 읽으면 0
        assert (await repo.refresh_judgments(older, now=NOW)).judged == 1
        assert (await repo.refresh_judgments(older, now=NOW)).judged == 0
        tomorrow = await repo.refresh_judgments(older, now=NOW + timedelta(days=1))
        assert tomorrow.judged == 3  # 신청 기간이 끝나지 않은 청년정책은 날마다 다시
        # 끝난 정책(E)은 날이 바뀌어도 다시 판정하지 않는다: 처음 보는 L·F·Y만
        later = await repo.refresh_judgments(
            late, now=NOW + timedelta(days=1), include_expired=True
        )
        assert later.judged == 3
    finally:
        await cleanup(conn, log_conn)
```
