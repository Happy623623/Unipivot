# 재수집이 준비하기가 만든 서류를 지우지 않게 (F-30 후속)

> 📝 기준: PRD v0.9 F-30 · ERD v0.9 · 2026-10-14
>
> 기준 커밋: main `68643aa`(#27 준비하기). `준비하기_구현.md`의 "남은 것" 두 번째 항목을 고친다

준비하기(#27)를 합치고 보니, 공고를 **다시 수집하면 준비하기가 만들어 넣은 서류가 지워진다**. 청년정책을 준비한 학생은 정책 내용이 바뀌어 다시 받는 순간 "필요 서류"가 비고, 할 일은 남는데 서류 연결이 끊긴다. 청년정책은 정기 수집 대상이라 켜기 전에 고쳐야 한다.

같은 자리에 하나 더 있었다. 학교 공지를 다시 추출하면 서류를 통째로 지웠다 넣어서, **준비하기가 모델에 물어 채운 안내(발급처·방법·소요 일수)도 함께 사라졌다.** 다음 학생이 그 공고를 준비하면 같은 서류를 또 묻는다(LLM 비용은 최대한 들이지 않는다 — 핵심 결정).

**규모와 확인:**
- api 파일 3개만 바뀐다(`app/extraction/store.py`와 테스트 2개). 새 파일이 없다.
- **DB 변경이 없다.** 합친 뒤 db push가 필요 없다. 기준 문서는 ERD 한 곳만 고친다.
- api `ruff check`·`ruff format --check` 통과, `552 passed, 71 skipped`(DB 없이) · `623 passed`(새 DB). web은 바뀌지 않는다.
- 되돌림 시험(변이) 4가지를 모두 테스트가 잡았다(아래 "확인한 것").
- **이 PR을 `로그연결_구현.md`(실행 로그 연결)보다 먼저 합친다.** 그 PR이 여기서 고치는 테스트 파일 2개(`test_extraction_db.py`·`test_youth_import.py`)를 함께 건드려서, 그 문서의 코드는 이 PR이 들어간 상태를 전제로 쓰여 있다.

## 남은 것

- **요건은 그대로 지우고 다시 넣는다.** 지금은 잃을 것이 없지만, 나중에 요건 행을 가리키는 데이터(예: 학생이 "이 조건은 내가 확인했다"고 표시)가 생기면 서류와 같은 문제가 난다.
- **서류 이름이 바뀌면 새 서류로 본다.** "재학증명서" → "재학 증명서"처럼 글자만 달라져도 안내를 다시 묻는다. 준비하기의 이름 정규화(`app/prepare/documents.py`의 `normalize`)를 여기서도 쓸 수 있지만, 공고 화면에 보이는 이름이 바뀌는 문제라 따로 본다.
- **할 일이 걸려 남은 서류는 공고에서 빠져도 계속 보인다.** 그 공고를 처음 여는 학생에게도 보인다. 준비한 학생 수가 적은 지금은 괜찮고, 서류마다 "지금 공고에 있는지"를 표시하려면 컬럼이 필요하다(ERD 변경).

  공고 상세가 그대로 보여 준다(F-42).

from collections.abc import Iterable, Sequence

def content_hash(body: str, attachments: Iterable[AttachmentInput]) -> str:
    """공고가 바뀌었는지 보는 값: 본문 글자(공백 정리) + 첨부 파일 sha256(첨부 순서대로).

    첨부는 추출 글자가 아니라 파일 해시를 쓴다. 이미지·스캔본은 Vision 전에는 글자가 없고,
    해시를 내려고 첨부를 모두 읽으면 비용이 든다.
    """
    digest = hashlib.sha256(" ".join(body.split()).encode())
    for attachment in sorted(attachments, key=lambda item: item.seq):
        digest.update(b"\n" + hashlib.sha256(attachment.data).hexdigest().encode())
    return digest.hexdigest()

async def save_extraction(
    conn: AsyncConnection,
    opportunity_id: str,
    result: ExtractionResult,
    *,
    run_id: str | None,
    content_hash: str,
) -> int:
    """저장하고 공고의 requirements_version을 돌려준다."""
    async with conn.transaction():
        row = await (
            await conn.execute(
                "select requirements_version, extraction_run_id from opportunities"
                " where id = %s for update",
                (opportunity_id,),
            )
        ).fetchone()
        if row is None:
            raise LookupError(f"공고가 없어요: {opportunity_id}")
        version: int = row["requirements_version"]
        if row["extraction_run_id"] is not None:
            version += 1  # 지금 요건이 바뀐다(새 요건이든, 실패로 비우든)
        await _replace_children(conn, opportunity_id, result)
        if result.succeeded:
            await conn.execute(
                "update opportunities set apply_start_at = %(apply_start_at)s,"
                " deadline_at = %(deadline_at)s, eligibility_basis_date = %(basis_date)s,"
                " easy_summary = %(summary)s, extraction_confidence = %(confidence)s,"
                " needs_review = %(needs_review)s, review_reasons = %(reasons)s,"
                " extraction_run_id = %(run_id)s, content_hash = %(hash)s,"
                " requirements_version = %(version)s where id = %(id)s",
                {
                    "apply_start_at": result.apply_start_at,
                    "deadline_at": result.deadline_at,
                    "basis_date": result.eligibility_basis_date,
                    "summary": result.easy_summary,
                    "confidence": result.confidence,
                    "needs_review": result.needs_review,
                    "reasons": list(result.review_reasons),
                    "run_id": run_id,
                    "hash": content_hash,
                    "version": version,
                    "id": opportunity_id,
                },
            )
        else:
            await conn.execute(
                "update opportunities set needs_review = true, review_reasons = %s,"
                " extraction_run_id = null, extraction_confidence = null, content_hash = %s,"
                " requirements_version = %s where id = %s",
                (list(result.review_reasons), content_hash, version, opportunity_id),
            )
        for attachment in result.attachments:
            if attachment.id is None or (attachment.method is None and attachment.error is None):
                continue  # 읽지 않은 첨부는 그대로 둔다(extract_method null)
            await conn.execute(
                "update opportunity_attachments set extracted_text = %s, extract_method = %s,"
                " extract_error = %s, extracted_at = now(), sha256 = coalesce(sha256, %s),"
                " mime_type = coalesce(mime_type, %s) where id = %s and opportunity_id = %s",
                (
                    attachment.text,
                    attachment.method,
                    attachment.error,
                    attachment.sha256,
                    attachment.mime_type,
                    attachment.id,
                    opportunity_id,
                ),
            )
    return version


async def _replace_children(
    conn: AsyncConnection, opportunity_id: str, result: ExtractionResult
) -> None:
    await conn.execute("delete from requirements where opportunity_id = %s", (opportunity_id,))
    # 요건은 clause_no, created_at 순으로 읽는다(판정, 상세 판정표). 한 트랜잭션의 now()는 모두 같아
    # 그대로 두면 묶음 안 순서가 id(무작위)로 정해진다. 1마이크로초씩 늘려 제출한 순서를 남긴다
    for index, requirement in enumerate(result.requirements):
        await conn.execute(
            "insert into requirements (opportunity_id, clause_no, field, operator, value, basis,"
            " evidence_text, is_ambiguous, created_at) values (%s, %s, %s, %s::req_operator, %s,"
            " %s, %s, %s, now() + %s * interval '1 microsecond')",
            (
                opportunity_id,
                requirement.clause_no,
                requirement.field,
                requirement.operator,
                Jsonb(requirement.value),
                requirement.basis,
                requirement.evidence_text,
                requirement.is_ambiguous,
                index,
            ),
        )
    await _merge_documents(conn, opportunity_id, result.documents)


async def _merge_documents(
    conn: AsyncConnection, opportunity_id: str, documents: Sequence[DocumentDraft]
) -> None:
    """서류는 지웠다 다시 넣지 않고 이름으로 맞춰 갱신한다.

    서류 행에는 수집이 모르는 값이 붙는다. 준비하기가 채운 안내(issuer·how_to·lead_days·effort_minutes,
    사전에 없는 서류는 모델에 물어 채운다)와 학생의 플래너 할 일 연결(planner_tasks.document_id)이다.
    지웠다 다시 넣으면 이미 준비한 학생의 서류 체크가 끊기고 안내도 다시 물어야 한다.

    - 새 목록이 비면 아무것도 건드리지 않는다. 청년정책은 수집이 서류를 주지 않고(제출 서류는 글이라
      준비하기가 만들어 넣는다), 학교 공지도 서류를 못 찾았다고 이미 있는 서류를 지울 이유는 없다
    - 이름이 같은 서류는 그대로 두고 is_required·form_url만 맞춘다
    - 새 목록에 없는 서류는 지우되, 누가 준비해 할 일이 걸린 서류는 남긴다(그 학생 화면이 비지 않게)
    """
    if not documents:
        return
    rows = await (
        await conn.execute(
            "select id, name from opportunity_documents where opportunity_id = %s",
            (opportunity_id,),
        )
    ).fetchall()
    wanted: dict[str, DocumentDraft] = {}
    for document in documents:
        wanted.setdefault(document.name, document)  # 같은 이름이 두 번 오면 처음 것만 쓴다
    kept: set[str] = set()
    for row in rows:
        name, document_id = row["name"], str(row["id"])
        if name in wanted and name not in kept:
            kept.add(name)
            await conn.execute(
                "update opportunity_documents set is_required = %s, form_url = %s where id = %s",
                (wanted[name].is_required, wanted[name].form_url, document_id),
            )
            continue
        await conn.execute(
            "delete from opportunity_documents where id = %s"
            " and not exists (select 1 from planner_tasks t where t.document_id = %s)",
            (document_id, document_id),
        )
    for name, document in wanted.items():
        if name in kept:
            continue
        await conn.execute(
            "insert into opportunity_documents (opportunity_id, name, is_required, form_url)"
            " values (%s, %s, %s, %s)",
            (opportunity_id, name, document.is_required, document.form_url),
        )

async def test_age_is_judged_on_the_days_one_can_apply() -> None:
    """청년정책은 마감일 하나로 나이를 따지지 않는다(판정엔진_규칙.md 8장). 신청할 수 있는 첫날(오늘, 신청 기간
    전이면 시작일)과 마감일 중 맞는 날이 있으면 그 결과다. 끝난 정책은 마감일이다. 날이 바뀌면 다시 판정한다."""
    conn, log_conn = await connect(), await connect(autocommit=True)
    try:
        await skip_if_other_youth_policies(conn, PREFIX)
        older = await make_user(conn, birth_date="1992-01-01")  # 오늘 만 34세
        freshman = await make_user(conn, birth_date="2007-11-15")  # 오늘 만 18세, 11/15에 19세
        late = await make_user(conn, birth_date="1991-10-01")  # 9월에 만 34세, 10/1부터 35세
        teen = await make_user(conn, birth_date="2007-10-05")  # 9월에 만 18세, 10/5부터 19세
        ages = {"sprtTrgtMinAge": "19", "sprtTrgtMaxAge": "34", "aplyPrdSeCd": "0057001"}
        await run(
            conn,
            log_conn,
            item("E", **ages, aplyYmd="20260901 ~ 20260930"),
            today=date(2026, 9, 15),
        )
        await run(
            conn,
            log_conn,
            item("L", **ages, aplyYmd="20260101 ~ 20281231"),  # 몇 년짜리 신청 기간
            item("F", **ages, aplyYmd="20270101 ~ 20270131"),  # 아직 시작 전
            item("Y", **ages, aplyYmd="20261001 ~ 20261117"),  # 마감 전에 19세가 된다
            item("E", **ages, aplyYmd="20260901 ~ 20260930"),  # 끝남 → 만료
        )
        saved = await rows(conn)
        repo = OpportunityRepository(conn)

        async def judged(user: str, name: str) -> tuple[date | None, str]:
            found = (await repo.detail(user, str(saved[name]["id"]), now=NOW)).eligibility
            return found.basis_date, found.display_status

        # 오늘 만 34세라 지원 가능. 마감일(2028-12-31)로만 따지면 만 36세라 지원 어려움이 된다
        assert await judged(older, "L") == (TODAY, "eligible")
        assert await judged(older, "F") == (date(2027, 1, 1), "ineligible")  # 시작일에 만 35세
        # 오늘은 만 18세지만 마감일엔 19세: 지원 가능, 기준일은 마감일
        assert await judged(freshman, "Y") == (date(2026, 11, 17), "eligible")
        # 끝난 정책은 마감일(9/30, 만 34세)로 본다. 오늘(만 35세)로 다시 따지지 않는다
        assert saved["E"]["status"] == "expired"
        assert await judged(late, "E") == (date(2026, 9, 30), "eligible")
        assert await judged(teen, "E") == (date(2026, 9, 30), "ineligible")  # 오늘 19세여도

        # 상세에서 판정한 L·F는 그대로, 처음 보는 Y만 판정한다. 같은 날 다시 읽으면 0
        assert (await repo.refresh_judgments(older, now=NOW)).judged == 1
        assert (await repo.refresh_judgments(older, now=NOW)).judged == 0
        tomorrow = await repo.refresh_judgments(older, now=NOW + timedelta(days=1))
        assert tomorrow.judged == 3  # 신청 기간이 끝나지 않은 청년정책은 날마다 다시
        # 끝난 정책(E)은 날이 바뀌어도 다시 판정하지 않는다: 처음 보는 L·F·Y만
        later = await repo.refresh_judgments(
            late, now=NOW + timedelta(days=1), include_expired=True
        )
        assert later.judged == 3
    finally:
        await cleanup(conn, log_conn)
